This Python calculation reproduces the post hoc prefix comparison from the repository root. The code below uses the late signature stored in each attempt. The referee separately regenerated every such signature from its completed rounds and compared every accepted event's two literal traces with its two source attempts.

```python
from collections import Counter, defaultdict
import gzip
import json
from pathlib import Path

folder = Path('artifacts/GROWTH-STATISTICS-20261121-001B/GROWTH-STATISTICS-20261121-001B-E001_search')
reference = json.loads((folder / 'calibration.json').read_bytes())['signature_dictionary']
totals = Counter()
comparisons = defaultdict(Counter)
with gzip.open(folder / 'search-results.jsonl.gz', 'rt') as stream:
    for line in stream:
        repeat = json.loads(line)['result']
        law, arms = repeat['law'], repeat['arms']
        known = set(map(tuple, reference[law]))
        limit = min(arm['evaluated_founders'] for arm in arms.values())
        scores = {}
        for name, arm in arms.items():
            grown = [row for row in arm['rows'] if row['eligible'] and 'growth' in row][:limit]
            end = grown[-1]['index'] if grown else -1
            seen = Counter(tuple(row['late_signature']) for row in grown if row['late_signature'] is not None)
            repeated = {key for key, count in seen.items() if count >= 2 and key not in known}
            checked = {tuple(event['key']) for event in arm['verification']['events']
                       if event['kind'] == 'novel' and event['status'] == 'verified'
                       and max(event['attempts']) <= end}
            assert repeated == checked
            scores[name] = len(checked)
            totals[law, name] += len(checked)
        difference = scores['diversity'] - scores['independent']
        comparisons[law][(difference > 0) - (difference < 0)] += 1
print(totals, dict(comparisons))
```
